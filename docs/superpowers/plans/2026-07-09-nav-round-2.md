import type { GeofenceAlertPayload } from './stationPauseLogic';

/** Zone types that should surface a transient HUD notification on enter
 *  (station handled separately by stationPauseAction; these are informational). */
const ALERTABLE_ZONE_TYPES = new Set(['alert', 'patrol_required']);

export interface ZoneAlertResult {
  show: boolean;
  zoneType: string;
}

export function zoneEntryAlert(payload: GeofenceAlertPayload, zoneName: string | undefined): ZoneAlertResult | null {
  if (payload.eventType !== 'enter') return null;
  if (!ALERTABLE_ZONE_TYPES.has(payload.zoneType)) return null;
  return { show: true, zoneType: payload.zoneType };
}

export interface WeatherSnapshot {
  precipitationType: string | null; // e.g. 'rain' | 'snow' | 'freezing_rain' | null
  windSpeedMph: number | null;
  visibilityMi: number | null;
}

export function weatherHazardLabel(w: WeatherSnapshot): string | null {
  if (w.precipitationType === 'freezing_rain' || w.precipitationType === 'ice') return 'Icy conditions';
  if (w.precipitationType === 'snow') return 'Snow';
  if (w.windSpeedMph != null && w.windSpeedMph >= 35) return 'High wind';
  if (w.visibilityMi != null && w.visibilityMi < 1) return 'Low visibility';
  return null;
}

```ts
// client/src/pages/navigation/hud/voiceGuidance.ts
export interface Maneuver {
  instruction: string; // e.g. "Turn right on Main Street"
  distanceMetersRemaining: number;
}

const ANNOUNCE_THRESHOLDS_M = [1609, 805, 402, 30]; // 1mi, 0.5mi, 0.25mi, "now"

/** Given the current maneuver and which thresholds have already been
 *  announced for it, decide whether to announce now and at which threshold
 *  (so the caller can mark it announced and avoid repeating). */
export function nextAnnouncement(
  maneuver: Maneuver,
  alreadyAnnounced: Set<number>,
): { thresholdM: number; text: string } | null {
  for (const t of ANNOUNCE_THRESHOLDS_M) {
    if (maneuver.distanceMetersRemaining <= t && !alreadyAnnounced.has(t)) {
      const distancePhrase = t === 30 ? 'now' : t === 402 ? 'in a quarter mile' : t === 805 ? 'in half a mile' : 'in one mile';
      return { thresholdM: t, text: `${maneuver.instruction} ${distancePhrase}` };
    }
  }
  return null;
}
```
